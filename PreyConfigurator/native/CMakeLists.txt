cmake_minimum_required(VERSION 3.20)
project(PreySettingsNative LANGUAGES CXX RC)
add_executable(PreySettings WIN32 main.cpp settings.cpp app.rc)
target_compile_features(PreySettings PRIVATE cxx_std_17)
target_compile_definitions(PreySettings PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PreySettings PRIVATE comctl32 shell32 user32 gdi32)
set_target_properties(PreySettings PROPERTIES OUTPUT_NAME "Prey2006 Reawakened Launcher" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(PreySettings PRIVATE /utf-8 /W4 /permissive- /Gy)
  target_link_options(PreySettings PRIVATE /OPT:REF /OPT:ICF /MANIFEST:NO)
endif()

# Development-only process probe, never distributed with the launcher.
add_executable(PreySettingsTestChild WIN32 test_child.cpp settings.cpp)
target_compile_features(PreySettingsTestChild PRIVATE cxx_std_17)
target_compile_definitions(PreySettingsTestChild PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(PreySettingsTestChild PRIVATE shell32 user32)
set_target_properties(PreySettingsTestChild PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(PreySettingsTestChild PRIVATE /utf-8 /W4)
endif()
